## Goal
Add a metadata preview step before YouTube upload so the operator can confirm
title, description, duration, and thumbnail are correct — without downloading the
video. Uses `yt-dlp --dump-json` (fast, ~2 seconds, zero bytes downloaded).

---

## Prerequisites
Phases 1–14 complete. `vidget publish` command exists and dispatches a GitHub
Actions workflow for YouTube upload.

---

## Design decision: why metadata-only
Full video preview via temporary cloud storage (Cloudflare R2, S3, etc.) adds
4 new credentials, a new module, egress concerns, and a staging/deletion lifecycle
for a problem that metadata solves well:

- Thumbnail confirms it's the right video visually
- Title + description confirm the upload metadata is correct
- Duration confirms the right cut/version was selected

If you need frame-level quality review before upload, do a local download first
(`vidget download <url>`) and watch it before triggering `publish`.

### 1. `VideoPreview` model — update `src/jre_vidget/models.py`

Add after the existing models:

```python
class VideoPreview(BaseModel):
    """Metadata fetched before download — used to confirm before upload."""

    url: str
    title: str
    description: str
    duration_seconds: int
    thumbnail_url: str
    uploader: str
    channel_url: str | None = None
    view_count: int | None = None
    upload_date: str | None = None   # YYYYMMDD string from yt-dlp
    formats: list[str] = Field(default_factory=list)  # e.g. ["1080p60", "720p", "480p"]

    @property
    def duration_display(self) -> str:
        """Return HH:MM:SS or MM:SS string."""
        h, rem = divmod(self.duration_seconds, 3600)
        m, s = divmod(rem, 60)
        if h:
            return f"{h}:{m:02d}:{s:02d}"
        return f"{m}:{s:02d}"
```

### 2. `engine.preview()` — update `src/jre_vidget/engine.py`

Add a new function (do NOT modify `download()`):

```python
def preview(url: str) -> VideoPreview:
    """
    Fetch video metadata without downloading any media.
    Raises DownloadError on network failure or unsupported URL.
    """
    import yt_dlp

    ydl_opts = {
        "quiet": True,
        "no_warnings": True,
        "skip_download": True,
    }

    try:
        with yt_dlp.YoutubeDL(ydl_opts) as ydl:
            info = ydl.extract_info(url, download=False)
    except yt_dlp.utils.DownloadError as exc:
        raise DownloadError(str(exc)) from exc

    if info is None:
        raise DownloadError(f"No metadata returned for {url}")

    # Collect available format labels (resolution strings)
    formats: list[str] = []
    for f in info.get("formats", []):
        label = f.get("format_note") or f.get("resolution")
        if label and label not in formats:
            formats.append(label)

    return VideoPreview(
        url=url,
        title=info.get("title", ""),
        description=info.get("description", ""),
        duration_seconds=int(info.get("duration") or 0),
        thumbnail_url=info.get("thumbnail", ""),
        uploader=info.get("uploader", ""),
        channel_url=info.get("channel_url"),
        view_count=info.get("view_count"),
        upload_date=info.get("upload_date"),
        formats=formats,
    )
```

```python
@app.command()
def preview(
    url: str = typer.Argument(..., help="Video URL to preview"),
    json_output: bool = typer.Option(
        False, "--json", help="Output raw JSON (for scripting)"
    ),
):
    """Fetch and display video metadata without downloading."""
    try:
        meta = engine.preview(url)
    except DownloadError as exc:
        ui.print_error("Preview failed", str(exc))
        raise typer.Exit(1)

    if json_output:
        import json
        typer.echo(json.dumps(meta.model_dump(), indent=2))
        return

    ui.print_preview(meta)
```

### 4. `ui.print_preview()` — update `src/jre_vidget/ui.py`

Add after existing print functions:

```python
def print_preview(meta: "VideoPreview") -> None:
    """Render a Rich preview card for a VideoPreview."""
    from rich.table import Table
    from rich.panel import Panel

    table = Table(show_header=False, box=None, padding=(0, 1))
    table.add_column("Field", style="bold cyan", width=14)
    table.add_column("Value")

    table.add_row("Title", meta.title)
    table.add_row("Uploader", meta.uploader)
    table.add_row("Duration", meta.duration_display)
    if meta.view_count is not None:
        table.add_row("Views", f"{meta.view_count:,}")
    if meta.upload_date:
        d = meta.upload_date
        table.add_row("Uploaded", f"{d[:4]}-{d[4:6]}-{d[6:]}")
    if meta.formats:
        table.add_row("Formats", ", ".join(meta.formats[:6]))
    table.add_row("URL", meta.url)

    console.print(Panel(table, title="[bold]Video Preview[/bold]", border_style="blue"))

    if meta.thumbnail_url:
        console.print(f"\n[dim]Thumbnail:[/dim] {meta.thumbnail_url}")
```

### 5. Two-step publish flow — update `src/jre_vidget/cli.py`

Modify the existing `publish` command to show a preview and prompt for confirmation
before dispatching the GitHub Actions workflow:

```python
@app.command()
def publish(
    url: str = typer.Argument(..., help="Video URL to publish to YouTube"),
    title: str | None = typer.Option(None, "--title", "-t", help="Override video title"),
    description: str | None = typer.Option(None, "--description", "-d"),
    yes: bool = typer.Option(False, "--yes", "-y", help="Skip confirmation prompt"),
):
    """Preview video metadata, then dispatch the YouTube upload workflow."""

    # Step 1: fetch metadata
    console.print("[bold cyan]Fetching metadata...[/bold cyan]")
    try:
        meta = engine.preview(url)
    except DownloadError as exc:
        ui.print_error("Preview failed — cannot confirm before upload", str(exc))
        raise typer.Exit(1)

    # Apply overrides
    if title:
        meta = meta.model_copy(update={"title": title})
    if description:
        meta = meta.model_copy(update={"description": description})

    # Step 2: show preview
    ui.print_preview(meta)

    # Step 3: confirm
    if not yes:
        confirmed = typer.confirm("\nPublish this video to YouTube?", default=False)
        if not confirmed:
            console.print("[yellow]Publish cancelled.[/yellow]")
            raise typer.Exit(0)

    # Step 4: dispatch workflow (existing logic)
    _dispatch_publish_workflow(url=url, title=meta.title, description=meta.description)
```

The existing `_dispatch_publish_workflow()` helper (or inline workflow dispatch code) is
unchanged — this phase only adds the preview + confirmation gate before it runs.

### 6. Unit tests — `tests/unit/test_preview.py`

```python
import pytest
from unittest.mock import patch, MagicMock
from jre_vidget.engine import preview
from jre_vidget.models import VideoPreview, DownloadError


FAKE_INFO = {
    "title": "JRE #1234 — Guest Name",
    "description": "Episode description here.",
    "duration": 7320,
    "thumbnail": "https://i.ytimg.com/vi/abc123/maxresdefault.jpg",
    "uploader": "PowerfulJRE",
    "channel_url": "https://www.youtube.com/@PowerfulJRE",
    "view_count": 1_500_000,
    "upload_date": "20260101",
    "formats": [
        {"format_note": "1080p60"},
        {"format_note": "720p"},
        {"format_note": "480p"},
        {"format_note": "1080p60"},   # duplicate — should be deduplicated
    ],
}


def _make_mock_ydl(info: dict | None = None, raises: Exception | None = None):
    mock_ydl = MagicMock()
    if raises:
        mock_ydl.__enter__.return_value.extract_info.side_effect = raises
    else:
        mock_ydl.__enter__.return_value.extract_info.return_value = info
    mock_ydl.__exit__.return_value = False
    return mock_ydl


def test_preview_returns_video_preview():
    with patch("yt_dlp.YoutubeDL", return_value=_make_mock_ydl(FAKE_INFO)):
        result = preview("https://youtube.com/watch?v=abc123")

    assert isinstance(result, VideoPreview)
    assert result.title == "JRE #1234 — Guest Name"
    assert result.duration_seconds == 7320
    assert result.uploader == "PowerfulJRE"


def test_preview_duration_display_hours():
    with patch("yt_dlp.YoutubeDL", return_value=_make_mock_ydl(FAKE_INFO)):
        result = preview("https://youtube.com/watch?v=abc123")
    assert result.duration_display == "2:02:00"


def test_preview_deduplicates_formats():
    with patch("yt_dlp.YoutubeDL", return_value=_make_mock_ydl(FAKE_INFO)):
        result = preview("https://youtube.com/watch?v=abc123")
    assert result.formats.count("1080p60") == 1
    assert "720p" in result.formats
    assert "480p" in result.formats


def test_preview_raises_on_download_error():
    import yt_dlp
    with patch(
        "yt_dlp.YoutubeDL",
        return_value=_make_mock_ydl(raises=yt_dlp.utils.DownloadError("not found")),
    ):
        with pytest.raises(DownloadError):
            preview("https://bad.url/video")


def test_preview_raises_when_info_is_none():
    with patch("yt_dlp.YoutubeDL", return_value=_make_mock_ydl(info=None)):
        with pytest.raises(DownloadError, match="No metadata returned"):
            preview("https://youtube.com/watch?v=abc123")


def test_preview_handles_missing_optional_fields():
    minimal_info = {
        "title": "Short Video",
        "description": "",
        "duration": 45,
        "thumbnail": "",
        "uploader": "SomeChannel",
        "formats": [],
    }
    with patch("yt_dlp.YoutubeDL", return_value=_make_mock_ydl(minimal_info)):
        result = preview("https://youtube.com/watch?v=xyz")
    assert result.duration_display == "0:45"
    assert result.view_count is None
    assert result.channel_url is None
    assert result.formats == []
```

### 7. CLI integration test — `tests/unit/test_cli_preview.py`

```python
import pytest
from typer.testing import CliRunner
from unittest.mock import patch
from jre_vidget.cli import app
from jre_vidget.models import VideoPreview

runner = CliRunner()

FAKE_PREVIEW = VideoPreview(
    url="https://youtube.com/watch?v=abc123",
    title="JRE #1234",
    description="An episode.",
    duration_seconds=3600,
    thumbnail_url="https://i.ytimg.com/vi/abc123/maxresdefault.jpg",
    uploader="PowerfulJRE",
)


def test_preview_command_renders_card():
    with patch("jre_vidget.cli.engine.preview", return_value=FAKE_PREVIEW):
        result = runner.invoke(app, ["preview", "https://youtube.com/watch?v=abc123"])
    assert result.exit_code == 0
    assert "JRE #1234" in result.output
    assert "PowerfulJRE" in result.output


def test_preview_command_json_flag():
    with patch("jre_vidget.cli.engine.preview", return_value=FAKE_PREVIEW):
        result = runner.invoke(app, ["preview", "--json", "https://youtube.com/watch?v=abc123"])
    assert result.exit_code == 0
    import json
    data = json.loads(result.output)
    assert data["title"] == "JRE #1234"


def test_preview_command_exits_1_on_error():
    from jre_vidget.models import DownloadError
    with patch("jre_vidget.cli.engine.preview", side_effect=DownloadError("bad url")):
        result = runner.invoke(app, ["preview", "https://bad.url"])
    assert result.exit_code == 1


def test_publish_shows_preview_before_dispatching(monkeypatch):
    dispatched = []

    def fake_dispatch(**kwargs):
        dispatched.append(kwargs)

    with patch("jre_vidget.cli.engine.preview", return_value=FAKE_PREVIEW):
        with patch("jre_vidget.cli._dispatch_publish_workflow", side_effect=fake_dispatch):
            result = runner.invoke(
                app,
                ["publish", "--yes", "https://youtube.com/watch?v=abc123"],
            )

    assert result.exit_code == 0
    assert "JRE #1234" in result.output   # preview was shown
    assert len(dispatched) == 1            # workflow was dispatched


def test_publish_cancelled_when_not_confirmed():
    with patch("jre_vidget.cli.engine.preview", return_value=FAKE_PREVIEW):
        # CliRunner sends empty input → confirm defaults to N
        result = runner.invoke(
            app,
            ["publish", "https://youtube.com/watch?v=abc123"],
            input="\n",
        )
    assert result.exit_code == 0
    assert "cancelled" in result.output.lower()
```

## Acceptance criteria
- `vidget preview <url>` prints a Rich preview card with title, uploader, duration,
  view count, upload date, and available formats
- `vidget preview --json <url>` outputs valid JSON matching `VideoPreview.model_dump()`
- `vidget publish <url>` shows the preview card and prompts for confirmation before
  dispatching the upload workflow
- `vidget publish --yes <url>` skips the confirmation prompt (CI / scripting use)
- `--title` and `--description` overrides are reflected in the preview card before
  confirmation
- `pytest tests/unit/test_preview.py tests/unit/test_cli_preview.py` — all tests pass
- `uv run ty check src/` — zero errors
- `uv run ruff check src/` — zero warnings
- No bare `print()` in `engine.py`, `models.py`, or `ui.py`
